**Task:** Health and damage system temporal invariants

**File:** `health_invariants.h`
```cpp
#ifndef HEALTH_INVARIANTS_H
#define HEALTH_INVARIANTS_H

#include <cstdint>

// Maximum health value
const uint32_t MaxHealth = 100;

// Entity health state
enum class HealthState { ALIVE, DEAD };

// Damage event delegate type
using DamageDelegate = void (*)(uint32_t damage);

// Health and damage subsystem interface
class HealthSubsystem {
public:
    // Initialize entity health to maximum value
    virtual void initHealth() = 0;

    // Apply damage to entity health within invulnerability window
    virtual bool applyDamage(uint32_t damage) = 0;

    // Clamp entity health between 0 and MaxHealth
    virtual uint32_t clampHealth(uint32_t health) const = 0;

    // Transition entity death state exactly once
    virtual void transitionDeathState() = 0;

    // Dispatch event delegate for damage application
    virtual void dispatchDamageEvent(DamageDelegate callback, uint32_t damage) = 0;
};

#endif  // HEALTH_INVARIANTS_H
```

```cpp
#include "health_invariants.h"

// Entity health state transition invariant
bool isDeathStateTransitionInvariant(HealthSubsystem& subsystem) {
    return subsystem.getHealth() == 0 && subsystem.getDeathState() == HealthState::DEAD;
}

// Damage application within invulnerability window invariant
bool isDamageApplicationInvariant(HealthSubsystem& subsystem, uint32_t damage) {
    // Assume invulnerability window duration is 1 second
    const uint32_t invulnerabilityWindow = 1000;

    return subsystem.getInvulnerabilityTimer() <= invulnerabilityWindow && subsystem.applyDamage(damage);
}

// Resource clamping invariant
bool isResourceClampingInvariant(HealthSubsystem& subsystem, uint32_t health) {
    return subsystem.clampHealth(health) >= 0 && subsystem.clampHealth(health) <= MaxHealth;
}

// Exactly-once death state transition invariant
bool isExactlyOnceDeathStateTransitionInvariant(HealthSubsystem& subsystem) {
    // Assume death state transition occurs exactly once
    const uint32_t maxTransitions = 1;

    return subsystem.getDeathStateTransitions() == maxTransitions && subsystem.transitionDeathState();
}

// Event dispatch sequencing invariant
bool isEventDispatchSequencingInvariant(HealthSubsystem& subsystem, DamageDelegate callback, uint32_t damage) {
    // Assume event dispatch sequence is linear
    const uint32_t maxSequenceLength = 10;

    return subsystem.dispatchDamageEvent(callback, damage) && subsystem.getEventDispatchSequence().size() <= maxSequenceLength;
}
```
