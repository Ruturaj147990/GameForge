```cpp
#include "ThirdPersonController.h"

namespace gameplay {

ThirdPersonController::ThirdPersonController()
    : m_characterTransform(nullptr)
    , m_camera(nullptr)
    , m_inputSystem(nullptr)
    , m_moveDirection(0, 0, 0)
    , m_isJumping(false) {}

ThirdPersonController::~ThirdPersonController() {}

void ThirdPersonController::update(float deltaTime) {
    // Get the character's transform and camera
    m_characterTransform = Transform::getEntity()->getComponent<Transform>();
    m_camera = Camera::getEntity()->getComponent<Camera>();

    // Handle movement input
    if (m_inputSystem->isKeyDown(InputKey::FORWARD)) {
        m_moveDirection += Vector3(0, 0, -1);
    }
    if (m_inputSystem->isKeyDown(InputKey::BACKWARD)) {
        m_moveDirection += Vector3(0, 0, 1);
    }
    if (m_inputSystem->isKeyDown(InputKey::LEFT)) {
        m_moveDirection += Vector3(-1, 0, 0);
    }
    if (m_inputSystem->isKeyDown(InputKey::RIGHT)) {
        m_moveDirection += Vector3(1, 0, 0);
    }

    // Clamp movement speed
    m_moveDirection = Vector3::clampLength(m_moveDirection, MAX_SPEED);

    // Update character position
    m_characterTransform->setPosition(m_characterTransform->getPosition() + m_moveDirection * deltaTime);

    // Handle camera rotation
    if (m_inputSystem->isKeyDown(InputKey::MOUSE_LEFT)) {
        float rotationSpeed = 100.0f;
        Vector2 mouseDelta = m_inputSystem->getMouseDelta();
        m_camera->rotate(Vector3(0, 1, 0), -mouseDelta.x * rotationSpeed);
        m_camera->rotate(Vector3(1, 0, 0), -mouseDelta.y * rotationSpeed);
    }

    // Handle jump input
    if (m_inputSystem->isKeyDown(InputKey::JUMP) && !m_isJumping) {
        m_characterTransform->applyForce(Vector3(0, JUMP_HEIGHT, 0));
        m_isJumping = true;
    }
}

} // namespace gameplay
```
