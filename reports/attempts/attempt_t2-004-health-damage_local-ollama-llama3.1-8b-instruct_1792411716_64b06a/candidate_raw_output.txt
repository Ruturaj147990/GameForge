void HealthComponent::on_update(entt::registry& registry, float delta) {
    auto health = registry.get<Health>(entt::null);
    if (health && !health->is_invulnerable()) {
        // Clamp current health to max health range
        health->current_health = std::max(0.0f, std::min(health->max_health, health->current_health));
    }
}

void HealthComponent::take_damage(float amount) {
    auto health = get<Health>();
    if (health && !health->is_invulnerable()) {
        // Apply damage to current health
        health->current_health -= amount;
        // Set invulnerability window start time
        health->invulnerability_window_start_time = entt::time();
        // Emit death event when health reaches zero
        if (health->current_health <= 0.0f) {
            emit_event<DeathEvent>(entt::null);
        }
    }
}

} // namespace game
