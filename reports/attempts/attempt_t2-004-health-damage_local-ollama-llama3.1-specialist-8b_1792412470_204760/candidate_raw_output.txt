#ifndef HEALTH_COMPONENT_H
#define HEALTH_COMPONENT_H

#include "entt.hpp"
#include "component_interface.h"

namespace game {

struct Health {
    float max_health = 100.0f;
    float current_health = 100.0f;
    float invulnerability_window_duration = 2.0f; // seconds
    float invulnerability_window_start_time = 0.0f;

    bool is_invulnerable() const { return invulnerability_window_start_time > 0.0f && entt::time() - invulnerability_window_start_time < invulnerability_window_duration; }
};

class HealthComponent : public ComponentInterface<Health> {
public:
    void on_update(entt::registry& registry, float delta) override;
    void take_damage(float amount);
};

} // namespace game

#endif // HEALTH_COMPONENT_H

#include "HealthComponent.h"

namespace game {

void HealthComponent::on_update(entt::registry& registry, float delta) {
    auto health = registry.get<Health>(entt::null);
    if (health && !health->is_invulnerable()) {
        // Clamp current health to max health range
        health->current_health = std::max(0.0f, std::min(health->max_health, health->current_health));
    }
}

void HealthComponent::take_damage(float amount) {
    auto health = get<Health>();
    if (health && !health->is_invulnerable()) {
        // Apply damage to current health
        health->current_health -= amount;
        // Set invulnerability window start time
        health->invulnerability_window_start_time = entt::time();
        // Emit death event when health reaches zero
        if (health->current_health <= 0.0f) {
            emit_event("death");
        }
    }
}

} // namespace game

void DamageSystem::on_update(entt::registry& registry, float delta) {
    auto entities = registry.view<Health>();
    for (auto entity : entities) {

        if (health && !health->is_invulnerable()) {

            health->current_health -= 10.0f; // example damage amount
