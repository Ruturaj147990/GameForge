#include "entt.hpp"
#include "components/Component.h"

namespace game {

struct Health {
    float maxHealth = 100.0f;
    float currentHealth = 100.0f;
    float invulnerabilityWindowDuration = 2.0f; // seconds
    float invulnerabilityWindowTimer = 0.0f;

    void reset() {
        currentHealth = maxHealth;
        invulnerabilityWindowTimer = 0.0f;
    }
};

class HealthComponent : public Component<Health> {
public:
    static constexpr auto healthId = "health";

    void onEntityCreated(Entity entity) override {
        get(entity).reset();
    }

    void onDamage(Entity entity, float damage) {
        if (get(entity).invulnerabilityWindowTimer > 0.0f) {
            // Invulnerability window is active
            return;
        }

        float newHealth = std::max(0.0f, get(entity).currentHealth - damage);
        get(entity).currentHealth = newHealth;

        if (newHealth <= 0.0f) {
            // Entity has died
            emitDeath(entity);
        } else {
            // Update invulnerability window timer
            get(entity).invulnerabilityWindowTimer = get(entity).invulnerabilityWindowDuration;
        }
    }

private:
    void emitDeath(Entity entity) {
        // Emit death event to the game world

    }
};

} // namespace game
