void PickupCollectionComponent::CollectItem(Item* item) const {
    if (m_collectedItem == nullptr && !IsCollected()) {
        m_collectedItem = item;
        m_inventory->IncrementItemCount(item);
        // Trigger UI update for inventory counter
        EntityManager::GetInstance()->GetUIManager()->UpdateInventoryCounter();
    }
}

    return m_collectedItem != nullptr;
}
