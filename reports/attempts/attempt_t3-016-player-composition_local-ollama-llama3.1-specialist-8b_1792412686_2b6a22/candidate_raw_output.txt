#include "PlayerPrefab/ActorComposition.h"
#include "GameplaySystem/GameplaySourcePaths.h"
#include "Scene/Prefab/ContentPaths.h"

namespace GameForge {

// Define a struct to hold the required references for the player prefab/actor composition.
struct PlayerCompositionReferences {
    Gameplay::EntityHandle gameplayEntity;
    Collision::ColliderHandle collisionCollider;
    Input::ControllerHandle inputController;
    Camera::CameraHandle camera;
    UI::CanvasHandle uiCanvas;
};

// Function to create or repair the designated player prefab/actor composition.
void createOrRepairPlayerComposition(PlayerCompositionReferences& references) {
    // Create a new gameplay entity if it doesn't exist.
    if (!references.gameplayEntity.isValid()) {
        references.gameplayEntity = Gameplay::createEntity(GameplaySourcePaths::getGameplaySourcePath());
    }

    // Add required components to the gameplay entity.
    references.gameplayEntity.addComponent<Gameplay::PlayerComponent>();
    references.gameplayEntity.addComponent<Collision::ColliderComponent>();

    // Create a new collision collider if it doesn't exist.
    if (!references.collisionCollider.isValid()) {
        references.collisionCollider = Collision::createCollider(CollisionSourcePaths::getCollisionSourcePath());
    }

    // Add the gameplay entity to the scene.
    Scene::addEntityToScene(references.gameplayEntity);

    // Create a new input controller if it doesn't exist.
    if (!references.inputController.isValid()) {
        references.inputController = Input::createController(InputSourcePaths::getInputSourcePath());
    }

    // Add the input controller to the gameplay entity.
    references.gameplayEntity.addComponent<Input::ControllerComponent>(references.inputController);

    // Create a new camera if it doesn't exist.
    if (!references.camera.isValid()) {
        references.camera = Camera::createCamera(CameraSourcePaths::getCameraSourcePath());
    }

    // Add the camera to the scene.
    Scene::addEntityToScene(references.camera);

    // Create a new UI canvas if it doesn't exist.
    if (!references.uiCanvas.isValid()) {
        references.uiCanvas = UI::createCanvas(UISourcePaths::getUIsourcePath());
    }

    // Add the UI canvas to the scene.
    Scene::addEntityToScene(references.uiCanvas);
}

}  // namespace GameForge
