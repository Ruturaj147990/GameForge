```cpp
#include "ThirdPersonController.h"

namespace gameplay {

ThirdPersonController::ThirdPersonController()
    : m_characterTransform(nullptr)
    , m_camera(nullptr)
    , m_inputSystem(nullptr)
    , m_moveDirection(0.0f, 0.0f, 0.0f)
    , m_isJumping(false) {
}

ThirdPersonController::~ThirdPersonController() {
}

void ThirdPersonController::update(float deltaTime) {
    // Get the character's transform and camera
    m_characterTransform = Transform::getEntity()->getComponent<Transform>();
    m_camera = Camera::getEntity()->getComponent<Camera>();

    // Handle movement input
    if (m_inputSystem->isKeyDown(InputKey::FORWARD)) {
        m_moveDirection += Vector3(0.0f, 0.0f, -1.0f);
    }
    if (m_inputSystem->isKeyDown(InputKey::BACKWARD)) {
        m_moveDirection += Vector3(0.0f, 0.0f, 1.0f);
    }
    if (m_inputSystem->isKeyDown(InputKey::LEFT)) {
        m_moveDirection += Vector3(-1.0f, 0.0f, 0.0f);
    }
    if (m_inputSystem->isKeyDown(InputKey::RIGHT)) {
        m_moveDirection += Vector3(1.0f, 0.0f, 0.0f);
    }

    // Clamp movement speed
    float speed = Math::clamp(m_moveDirection.length(), 0.0f, MAX_SPEED);

    // Update character position
    m_characterTransform->setPosition(
        m_characterTransform->getPosition() + (m_moveDirection * speed * deltaTime));

    // Handle jump input
    if (m_inputSystem->isKeyDown(InputKey::JUMP) && !m_isJumping) {
        m_isJumping = true;
        m_characterTransform->setVelocity(Vector3(0.0f, JUMP_HEIGHT, 0.0f));
    }

    // Update camera position
    Vector3 characterPosition = m_characterTransform->getPosition();
    Vector3 cameraPosition = characterPosition + (m_camera->getForward() * 5.0f);
    m_camera->setPosition(cameraPosition);

    // Reset movement direction
    m_moveDirection = Vector3(0.0f, 0.0f, 0.0f);
}

} // namespace gameplay
```
