#include "ProjectileActor.h"
#include "Core/Entity.h"
#include "Core/CollisionSystem.h"

namespace GameForge {

ProjectileActor::ProjectileActor(Entity* owner, float velocity)
    : m_owner(owner), m_velocity(velocity), m_isActive(true) {}

ProjectileActor::~ProjectileActor() {
    if (m_owner != nullptr && m_owner->GetHealth() > 0.0f) {
        m_owner->RemoveComponent(this);
    }
}

void ProjectileActor::Update(float deltaTime) {
    if (!m_isActive || m_owner == nullptr) return;

    // Update velocity and position
    m_owner->SetPosition(m_owner->GetPosition() + (m_velocity * deltaTime));

    // Check for collision with blocking geometry or entities
    CollisionSystem::GetInstance()->CheckCollision(m_owner);
}

void ProjectileActor::OnCollision(const Collision& collision) {
    if (!m_isActive || m_owner == nullptr) return;

    Entity* target = collision.GetTarget();
    if (target != nullptr && !target->IsInvulnerable()) {
        // Apply damage to the target
        float damage = m_velocity * 0.1f; // configurable damage multiplier
        target->ApplyDamage(damage);

        // Deactivate or destroy the projectile
        m_isActive = false;
        if (m_owner != nullptr) {
            m_owner->RemoveComponent(this);
        }
    } else if (target == nullptr || target->IsInvulnerable()) {
        // Handle invalid targets without errors
        m_isActive = false;
        if (m_owner != nullptr) {
            m_owner->RemoveComponent(this);
        }
    }
}

} // namespace GameForge
