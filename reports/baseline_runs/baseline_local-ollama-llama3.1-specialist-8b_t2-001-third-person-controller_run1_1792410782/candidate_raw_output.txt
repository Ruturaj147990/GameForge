#include "ThirdPersonController.h"
#include "gameplay/components/CameraOrbitComponent.h"

namespace gameplay {

const char* ThirdPersonController::NAME = "ThirdPersonController";

ThirdPersonController::ThirdPersonController()
    : m_characterTransform(nullptr)
    , m_camera(nullptr) {
}

ThirdPersonController::~ThirdPersonController() {
}

void ThirdPersonController::update(float deltaTime) {
    if (m_characterTransform && m_camera) {
        // Update character position
        Vector3 movement = getInputSystem()->getMovementInput();
        m_characterTransform->translate(movement * deltaTime);

        // Update camera orbit
        CameraOrbitComponent* orbit = m_camera->getCameraOrbitComponent();
        orbit->update(deltaTime);
    }
}

void ThirdPersonController::onMoveInput(const InputEvent& event) {
    getInputSystem()->setMovementInput(event.getVector3());
}

void ThirdPersonController::onLookInput(const InputEvent& event) {
    getInputSystem()->setLookInput(event.getVector2());
}

void ThirdPersonController::onJumpInput(const InputEvent& event) {
    if (event.isPressed()) {
        // Apply jump force
        m_characterTransform->applyForce(Vector3(0, 10, 0));
    }
}

} // namespace gameplay

#include "gameplay/ThirdPersonController.h"

namespace gameplay {

const char* CameraOrbitComponent::NAME = "CameraOrbitComponent";

CameraOrbitComponent::CameraOrbitComponent()
    : m_orbitSpeed(1.0f) {
}

CameraOrbitComponent::~CameraOrbitComponent() {
}

void CameraOrbitComponent::update(float deltaTime) {
    // Update camera position based on character position
    Transform* characterTransform = ThirdPersonController::get()->getCharacterTransform();
    Vector3 characterPosition = characterTransform->getPosition();

    // Calculate orbit angle
    float angle = (ThirdPersonController::get()->getLookInput().x * m_orbitSpeed) * deltaTime;

    // Update camera position
    Vector3 cameraPosition = characterPosition + Vector3(cos(angle), sin(angle), 0);
    ThirdPersonController::get()->getCamera()->setPosition(cameraPosition);
}

} // namespace gameplay
