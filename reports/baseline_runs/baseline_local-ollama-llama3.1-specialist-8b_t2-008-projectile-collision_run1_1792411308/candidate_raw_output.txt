    // Check if the collision is with a damageable target
    DamageableTarget* target = dynamic_cast<DamageableTarget*>(event.GetCollider());
    if (target != nullptr && !target->IsInvulnerable()) {
        ApplyDamage(target);
    }

    // Calculate the damage to apply based on projectile properties
    float damageToApply = m_Damage;
    target->TakeDamage(damageToApply);

    // Deactivate or destroy the projectile
    Entity* entity = GetEntity();
    if (entity != nullptr) {
        entity->SetActive(false);
