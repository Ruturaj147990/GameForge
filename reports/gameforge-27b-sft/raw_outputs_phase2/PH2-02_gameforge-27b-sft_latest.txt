// ACharacter.cpp
#include "ACharacter.h"
#include "Item.h"
#include "Engine/World.h"
#include "GameFramework/PlayerController.h"

ACharacter::ACharacter()
{
	PrimaryActorTick.bCanEverTick = false;
}

void ACharacter::AttemptPickup(AItem* Item)
{
	if (!Item)
	{
		return;
	}

	// Only allow local player to initiate pickup
	if (GetLocalRole() == ROLE_Authority)
	{
		// If we're the authority, call server directly
		Server_PickupItem(Item);
	}
	else
	{
		// Send to server
		Server_PickupItem(Item);
	}
}

bool ACharacter::Server_PickupItem_Validate(AItem* Item)
{
	// Validate item is not null
	if (!Item)
	{
		return false;
	}

	// Validate item is valid (not already picked up, etc.)
	if (!Item->IsValidForPickup())
	{
		return false;
	}

	// Validate distance
	const float MaxPickupDistance = 200.0f;
	const FVector Distance = GetActorLocation() - Item->GetActorLocation();
	if (Distance.SizeSquared() > FMath::Square(MaxPickupDistance))
	{
		return false;
	}

	return true;
}
