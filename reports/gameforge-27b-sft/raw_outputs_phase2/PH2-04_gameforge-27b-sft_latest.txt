// SensorComponent.cpp
#include "SensorComponent.h"

USensorComponent::USensorComponent()
{
	PrimaryComponentTick.bCanEverTick = false;
}

void USensorComponent::BeginPlay()
{
	Super::BeginPlay();

	if (GetWorld())
	{
		GetWorld()->GetTimerManager().SetTimer(
			SensorPollTimerHandle,
			this,
			&USensorComponent::PollSensors,
			0.2f,
			true
		);
	}
}

void USensorComponent::EndPlay(const EEndPlayReason::Type EndPlayReason)
{
	if (GetWorld())
	{
		GetWorld()->GetTimerManager().ClearTimer(SensorPollTimerHandle);
	}

	Super::EndPlay(EndPlayReason);
}
