{
	Super::Initialize(Collection);

	// Optionally load the default save game asset or create a new one
	// For now, we'll assume USaveGame is a UObject with UPROPERTYs for player data
	// In production, you'd load from a default asset or create a new instance
	CurrentSaveGame = NewObject<USaveGame>(this);
}

void UGameSaveSubsystem::Deinitialize()
{
	// Clean up any pending operations
	if (CurrentSaveGame)
	{
		CurrentSaveGame = nullptr;
	}

	Super::Deinitialize();
}
