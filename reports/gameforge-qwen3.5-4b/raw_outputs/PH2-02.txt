// ItemPickup.cpp
#include "ItemPickup.h"
#include "GameFramework/PlayerController.h"
#include "GameFramework/Authority.h"

#include "GameFramework/CharacterMovementComponent.h"

void AItemPickupCharacter::BeginPlay()
{
    Super::BeginPlay();
}

void AItemPickupCharacter::AttemptPickup(AItem* Item)
{
    // Validate Item Reference
    if (!Item || !Item->IsValidLowLevel())
    {
        UE_LOG(LogTemp, Error, TEXT("AttemptPickup: Invalid Item reference."));
        return;
    }

    // Validate Distance (e.g., 300 units)
    FVector PlayerLocation = GetActorLocation();
    FVector ItemLocation = Item->GetActorLocation();
    float Distance = FVector::Distance(PlayerLocation, ItemLocation);

    if (Distance > 300.0f)
    {
        UE_LOG(LogTemp, Warning, TEXT("AttemptPickup: Item is too far (%f > 300).", Distance));
        return;
    }

    // Invoke Server RPC
    Server_PickupItem(Item);
}
